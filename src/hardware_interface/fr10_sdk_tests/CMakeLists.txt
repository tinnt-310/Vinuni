cmake_minimum_required(VERSION 3.16)
project(fr10_sdk_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../fr10_driver/fairino_hardware_v3_9_3/libfairino" CACHE PATH "FAIRINO SDK root")
set(SDK_LIBRARY "${SDK_ROOT}/lib/libfairino.so.2.3.3" CACHE FILEPATH "Exact SDK library to test")
if(NOT EXISTS "${SDK_ROOT}/include/robot.h" OR NOT EXISTS "${SDK_LIBRARY}")
  message(FATAL_ERROR "Set SDK_ROOT and SDK_LIBRARY to matching headers and library")
endif()
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
find_package(Threads REQUIRED)
add_executable(sdk_probe native/sdk_probe.cpp)
target_include_directories(sdk_probe PRIVATE "${SDK_ROOT}/include" "${JSON_INCLUDE_DIR}")
target_link_libraries(sdk_probe PRIVATE "${SDK_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
get_filename_component(SDK_CANONICAL_LIBRARY "${SDK_LIBRARY}" REALPATH)
target_compile_definitions(sdk_probe PRIVATE FR10_EXPECTED_SDK="${SDK_CANONICAL_LIBRARY}")
get_filename_component(SDK_LIB_DIR "${SDK_LIBRARY}" DIRECTORY)
set_target_properties(sdk_probe PROPERTIES BUILD_RPATH "${SDK_LIB_DIR}")
target_link_options(sdk_probe PRIVATE "-Wl,--disable-new-dtags")
target_compile_options(sdk_probe PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(sdk_probe_fake native/sdk_probe.cpp tests/fake_sdk.cpp)
  target_include_directories(sdk_probe_fake PRIVATE "${SDK_ROOT}/include" "${JSON_INCLUDE_DIR}")
  target_link_libraries(sdk_probe_fake PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  target_compile_definitions(sdk_probe_fake PRIVATE FR10_OFFLINE_FAKE=1)
  target_compile_options(sdk_probe_fake PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME offline_contract COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_offline.py" --binary "$<TARGET_FILE:sdk_probe_fake>")
endif()
