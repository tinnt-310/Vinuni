cmake_minimum_required(VERSION 3.8)
project(charger_detection)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(message_filters REQUIRED)
find_package(ccs2_socket_detector REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(OpenCV REQUIRED)
add_executable(ccs2_detector_node src/ccs2_detector_node.cpp)
target_include_directories(ccs2_detector_node PRIVATE include)
ament_target_dependencies(ccs2_detector_node rclcpp sensor_msgs std_msgs cv_bridge OpenCV message_filters ccs2_socket_detector)
install(TARGETS ccs2_detector_node DESTINATION lib/${PROJECT_NAME})
add_executable(solvepnp src/solvepnp.cpp)
target_include_directories(solvepnp PRIVATE include)
ament_target_dependencies(solvepnp rclcpp sensor_msgs geometry_msgs cv_bridge OpenCV tf2 tf2_geometry_msgs)
install(TARGETS solvepnp DESTINATION lib/${PROJECT_NAME})
add_executable(depth_heatmap src/depth_heatmap.cpp)
target_include_directories(depth_heatmap PRIVATE include)
ament_target_dependencies(depth_heatmap rclcpp sensor_msgs cv_bridge OpenCV)
install(TARGETS depth_heatmap DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch/ DESTINATION share/${PROJECT_NAME}/launch)
ament_package()
