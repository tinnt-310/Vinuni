cmake_minimum_required(VERSION 3.8)
project(swerve_vehicle_control)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(controller_interface REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(mission_manager src/mission_manager_node.cpp)
ament_target_dependencies(mission_manager rclcpp rclcpp_action geometry_msgs nav2_msgs std_msgs tf2)
install(TARGETS mission_manager DESTINATION lib/${PROJECT_NAME})
add_library(swerve_controller_plugin SHARED src/swerve_controller_plugin.cpp)
target_include_directories(swerve_controller_plugin PRIVATE include)
target_link_libraries(swerve_controller_plugin Eigen3::Eigen)
ament_target_dependencies(swerve_controller_plugin rclcpp controller_interface realtime_tools pluginlib rclcpp_lifecycle builtin_interfaces geometry_msgs nav_msgs std_msgs tf2 tf2_ros lifecycle_msgs)
pluginlib_export_plugin_description_file(controller_interface controller_plugins.xml)
install(TARGETS swerve_controller_plugin DESTINATION lib)
install(PROGRAMS scripts/cmd_vel_normalizer.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch/ DESTINATION share/${PROJECT_NAME}/launch)
install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)
ament_package()
